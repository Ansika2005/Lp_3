import java.util.*;

public class FractionalKnapsack {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        System.out.print("Enter number of items: ");
        int n = sc.nextInt();

        double[] profit = new double[n];
        double[] weight = new double[n];
        double[] ratio = new double[n];
        int[] item = new int[n];

        for (int i = 0; i < n; i++) {
            item[i] = i + 1;

            System.out.print("Enter profit and weight of item " + (i + 1) + ": ");
            profit[i] = sc.nextDouble();
            weight[i] = sc.nextDouble();

            ratio[i] = profit[i] / weight[i];
        }

        System.out.print("Enter knapsack capacity: ");
        double capacity = sc.nextDouble();

        // Sort items by ratio in descending order
        for (int i = 0; i < n - 1; i++) {
            for (int j = i + 1; j < n; j++) {
                if (ratio[i] < ratio[j]) {
                    double temp;

                    temp = ratio[i];
                    ratio[i] = ratio[j];
                    ratio[j] = temp;

                    temp = profit[i];
                    profit[i] = profit[j];
                    profit[j] = temp;

                    temp = weight[i];
                    weight[i] = weight[j];
                    weight[j] = temp;

                    int t = item[i];
                    item[i] = item[j];
                    item[j] = t;
                }
            }
        }

        double totalProfit = 0;

        System.out.println("\nFractional Knapsack Table:");
        System.out.println("--------------------------------------------------------------------------");
        System.out.printf("%-8s %-10s %-10s %-10s %-12s %-12s %-10s%n",
                "Item", "Profit", "Weight", "Ratio",
                "Selected Wt.", "Selected %", "Earned");
        System.out.println("--------------------------------------------------------------------------");

        for (int i = 0; i < n; i++) {
            double selectedWeight;
            double earnedProfit;

            if (capacity >= weight[i]) {
                selectedWeight = weight[i];
                earnedProfit = profit[i];
                capacity -= weight[i];
            } else if (capacity > 0) {
                selectedWeight = capacity;
                earnedProfit = ratio[i] * capacity;
                capacity = 0;
            } else {
                selectedWeight = 0;
                earnedProfit = 0;
            }

            double percentage = (selectedWeight / weight[i]) * 100;

            totalProfit += earnedProfit;

            System.out.printf("%-8d %-10.2f %-10.2f %-10.2f %-12.2f %-11.2f%% %-10.2f%n",
                    item[i], profit[i], weight[i], ratio[i],
                    selectedWeight, percentage, earnedProfit);
        }

        System.out.println("--------------------------------------------------------------------------");
        System.out.printf("Maximum Profit: %.2f%n", totalProfit);

        sc.close();
    }
}
