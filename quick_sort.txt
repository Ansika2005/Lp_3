
import random
import time

# Deterministic Quick Sort
def deterministic_quick_sort(arr):

    if len(arr) <= 1:
        return arr

    pivot = arr[0]

    left = [x for x in arr[1:] if x <= pivot]
    right = [x for x in arr[1:] if x > pivot]

    return (deterministic_quick_sort(left)
            + [pivot]
            + deterministic_quick_sort(right))

# Randomized Quick Sort
def randomized_quick_sort(arr):

    if len(arr) <= 1:
        return arr

    pivot = random.choice(arr)

    left = [x for x in arr if x < pivot]
    right = [x for x in arr if x > pivot]

    equal = [x for x in arr if x == pivot]

    return (randomized_quick_sort(left)
            + equal
            + randomized_quick_sort(right))

# Input
n = int(input("Enter number of elements: "))

arr = list(map(int, input("Enter elements: ").split()))

# Deterministic Quick Sort Analysis
start = time.perf_counter()
result1 = deterministic_quick_sort(arr.copy())
end = time.perf_counter()

det_time = end - start

# Randomized Quick Sort Analysis
start = time.perf_counter()
result2 = randomized_quick_sort(arr.copy())
end = time.perf_counter()

rand_time = end - start

# Results
print("\nOriginal Array:", arr)

print("\nDeterministic Quick Sort:", result1)
print("Execution Time:", det_time, "seconds")

print("\nRandomized Quick Sort:", result2)
print("Execution Time:", rand_time, "seconds")
